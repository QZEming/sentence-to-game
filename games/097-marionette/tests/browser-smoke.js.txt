async (page) => {
 await page.reload();
 await page.waitForFunction(()=>!!document.getElementById("marionette-game").game?.getState().velocity);
 const read=()=>page.evaluate(()=>document.getElementById("marionette-game").game.getState());
 await page.getByRole("button",{name:"开始演出",exact:true}).click();
 await page.locator("canvas").click({position:{x:40,y:100}});
 await page.keyboard.down("w");await page.waitForTimeout(250);await page.keyboard.up("w");
 const moved=await read();
 await page.keyboard.press("p");const paused=await read();
 await page.keyboard.press("Space");await page.waitForTimeout(180);const still=await read();
 await page.getByRole("button",{name:"继续演出",exact:true}).click();await page.waitForTimeout(100);const resumed=await read();
 await page.keyboard.down("q");await page.waitForTimeout(140);
 const q=page.getByRole("button",{name:"Q · 左提线",exact:true});await q.hover();await page.mouse.down();await page.mouse.up();
 const mixedHeld=await q.evaluate(el=>el.classList.contains("held"));await page.keyboard.up("q");
 await page.getByRole("button",{name:"操作指南",exact:true}).click();const help=await read();
 await page.getByRole("button",{name:"回到舞台",exact:true}).click();const back=await read();
 await page.keyboard.press("p");
 const result={move:moved.position.z<4.8,pause:paused.status==="paused"&&still.simulationTime===paused.simulationTime,noBufferedJump:resumed.position.y===0,mixedInput:mixedHeld,help:help.status==="help"&&back.status==="playing",states:{moved,paused,still,resumed},overflow:await page.evaluate(()=>document.documentElement.scrollWidth>innerWidth)};
 if(!result.move||!result.pause||!result.noBufferedJump||!result.mixedInput||!result.help||result.overflow)throw Error(JSON.stringify(result));
 return result;
}
